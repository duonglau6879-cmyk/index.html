    // ================= PET ANIMATION CANVAS =================
    let canvas, ctx, petYOffset = 0, animDir = 1;

    function initPetCanvas() {
      canvas = document.getElementById('petCanvas');
      ctx = canvas.getContext('2d');
      animatePet();
    }

    function animatePet() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      // Bounce offset
      petYOffset += 0.3 * animDir;
      if (Math.abs(petYOffset) > 6) animDir *= -1;

      const currentPetType = friendsData[currentRoomIndex].pet.type;
      const centerX = canvas.width / 2;
      const centerY = canvas.height / 2 + petYOffset;

      // Draw Shadow
      ctx.beginPath();
      ctx.ellipse(centerX, centerY + 65, 50, 12, 0, 0, Math.PI * 2);
      ctx.fillStyle = 'rgba(0, 0, 0, 0.15)';
      ctx.fill();

      // Render Pet Graphic dynamically based on type
      if (currentPetType === 'dog') {
        // Dog Body
        ctx.beginPath();
        ctx.arc(centerX, centerY, 50, 0, Math.PI * 2);
        ctx.fillStyle = '#f59e0b';
        ctx.fill();
        // Dog Ears
        ctx.beginPath();
        ctx.ellipse(centerX - 40, centerY - 25, 15, 30, Math.PI / 6, 0, Math.PI * 2);
        ctx.ellipse(centerX + 40, centerY - 25, 15, 30, -Math.PI / 6, 0, Math.PI * 2);
        ctx.fillStyle = '#d97706';
        ctx.fill();
      } else if (currentPetType === 'cat') {
        // Cat Body
        ctx.beginPath();
        ctx.arc(centerX, centerY, 50, 0, Math.PI * 2);
        ctx.fillStyle = '#94a3b8';
        ctx.fill();
        // Ears
        ctx.beginPath();
        ctx.moveTo(centerX - 40, centerY - 20); ctx.lineTo(centerX - 20, centerY - 65); ctx.lineTo(centerX - 5, centerY - 40);
        ctx.moveTo(centerX + 40, centerY - 20); ctx.lineTo(centerX + 20, centerY - 65); ctx.lineTo(centerX + 5, centerY - 40);
        ctx.fillStyle = '#64748b';
        ctx.fill();
      } else if (currentPetType === 'dragon') {
        // Dragon Body
        ctx.beginPath();
        ctx.arc(centerX, centerY, 50, 0, Math.PI * 2);
        ctx.fillStyle = '#06b6d4';
        ctx.fill();
      } else {
        // Rabbit Body
        ctx.beginPath();
        ctx.arc(centerX, centerY, 45, 0, Math.PI * 2);
        ctx.fillStyle = '#e2e8f0';
        ctx.fill();
      }

      // Eyes
      ctx.beginPath();
      ctx.arc(centerX - 15, centerY - 10, 6, 0, Math.PI * 2);
      ctx.arc(centerX + 15, centerY - 10, 6, 0, Math.PI * 2);
      ctx.fillStyle = '#0f172a';
      ctx.fill();

      // Nose / Mouth
      ctx.beginPath();
      ctx.arc(centerX, centerY + 5, 4, 0, Math.PI * 2);
      ctx.fillStyle = '#f43f5e';
      ctx.fill();

      requestAnimationFrame(animatePet);
    }

      if (room.meat <= 0) {

      }
      room.meat -= 1;

      loadRoomData(currentRoomIndex);

    }

    // ================= MINI-GAMES ENGINE (6 GAMES) =================
    let currentGameId = null, gameLoop = null, gameScore = 0;

    function startGame(gameId) {
      currentGameId = gameId;
      gameScore = 0;
      document.getElementById('gamesList').classList.add('hidden');
      document.getElementById('activeGameArea').classList.remove('hidden');
      document.getElementById('gameScore').innerText = '0';

      const canvas = document.getElementById('miniGameCanvas');
      const ctx = canvas.getContext('2d');

      if (gameId === 1) {
        document.getElementById('gameTitle').innerText = '🧺 Game 1: Hứng Thịt Rơi';
        let basketX = 150, meatY = 0, meatX = Math.random() * 320;
        
        canvas.onmousemove = (e) => {
          const rect = canvas.getBoundingClientRect();
          basketX = e.clientX - rect.left - 25;
        };

        gameLoop = setInterval(() => {
          ctx.clearRect(0, 0, canvas.width, canvas.height);
          meatY += 4;

          // Draw Meat
          ctx.fillStyle = '#f59e0b';
          ctx.beginPath(); ctx.arc(meatX, meatY, 10, 0, Math.PI * 2); ctx.fill();

          // Draw Basket
          ctx.fillStyle = '#ec4899';
          ctx.fillRect(basketX, 190, 50, 15);

          // Collision Check
          if (meatY >= 180 && meatX >= basketX && meatX <= basketX + 50) {
            gameScore += 2;
            document.getElementById('gameScore').innerText = gameScore;
            meatY = 0; meatX = Math.random() * 320;
          } else if (meatY > 220) {
            meatY = 0; meatX = Math.random() * 320;
          }
        }, 20);
      } else {
        document.getElementById('gameTitle').innerText = '🎯 Game Mini-Interactive';
        ctx.fillStyle = '#f43f5e';
        ctx.font = '14px Arial';
        ctx.fillText('Chạm liên tục để tích điểm thịt!', 80, 110);
        canvas.onclick = () => {
          gameScore += 1;
          document.getElementById('gameScore').innerText = gameScore;
        };
      }
    }

    function quitGame() {
      clearInterval(gameLoop);
      if (gameScore > 0) {
        friendsData[currentRoomIndex].meat += gameScore;
        loadRoomData(currentRoomIndex);
      }
      document.getElementById('activeGameArea').classList.add('hidden');
      document.getElementById('gamesList').classList.remove('hidden');
      closeModal('gamesModal');
    }
